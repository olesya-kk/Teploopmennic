cmake_minimum_required(VERSION 3.16)
project(HeatExchanger LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets Sql Concurrent)
find_package(OpenMP REQUIRED)
set(SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/TEPLOOBMENNIK/ТЕПЛООБМЕННИК")
add_library(heatcore data/catalog.qrc ${SOURCE_DIR}/catalog.cpp ${SOURCE_DIR}/heatcalc.cpp ${SOURCE_DIR}/correlations.cpp ${SOURCE_DIR}/database.cpp)
target_include_directories(heatcore PUBLIC ${SOURCE_DIR})
target_link_libraries(heatcore PUBLIC Qt6::Sql OpenMP::OpenMP_CXX)
target_compile_options(heatcore PRIVATE -Wall -Wextra -Wpedantic)
add_executable(heat_exchanger ${SOURCE_DIR}/main.cpp ${SOURCE_DIR}/heatexchangerapp.cpp ${SOURCE_DIR}/heatexchangerapp.h)
target_link_libraries(heat_exchanger PRIVATE heatcore Qt6::Widgets Qt6::Concurrent)
enable_testing()
add_executable(heat_tests tests/core_tests.cpp)
target_link_libraries(heat_tests PRIVATE heatcore)
add_test(NAME core COMMAND heat_tests)
add_executable(heat_benchmark tests/benchmark.cpp)
target_link_libraries(heat_benchmark PRIVATE heatcore)
find_package(Qt6 REQUIRED COMPONENTS Test)
add_executable(heat_gui_tests tests/gui_tests.cpp ${SOURCE_DIR}/heatexchangerapp.cpp ${SOURCE_DIR}/heatexchangerapp.h)
target_link_libraries(heat_gui_tests PRIVATE heatcore Qt6::Widgets Qt6::Concurrent Qt6::Test)
add_test(NAME gui COMMAND heat_gui_tests)
set_tests_properties(gui PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 30)
